// --- RENDERIZADORES DE SELECTS ---
function renderGroupSelectors() {
  const groups = JSON.parse(localStorage.getItem("groups"));
  const globalSelector = document.getElementById("global-group-selector");
  globalSelector.innerHTML = "";
  
  groups.forEach(g => {
    const opt = document.createElement("option");
    opt.value = g.id;
    opt.textContent = g.nome;
    globalSelector.appendChild(opt);
  });

  if (currentGroup) {
    globalSelector.value = currentGroup.id;
  }
}

// --- CONFIGURAÇÃO DE MÁSCARAS E VALIDAÇÕES EM TEMPO REAL ---
function initFormMasks() {
  const cpfInputs = [document.getElementById("register-cpf"), document.getElementById("athlete-cpf")];
  const waInputs = [document.getElementById("register-whatsapp"), document.getElementById("athlete-whatsapp")];

  cpfInputs.forEach(input => {
    if (!input) return;
    input.addEventListener("input", (e) => {
      let v = e.target.value.replace(/\D/g, "");
      if (v.length > 11) v = v.slice(0, 11);
      if (v.length > 9) {
        v = `${v.slice(0, 3)}.${v.slice(3, 6)}.${v.slice(6, 9)}-${v.slice(9)}`;
      } else if (v.length > 6) {
        v = `${v.slice(0, 3)}.${v.slice(3, 6)}.${v.slice(6)}`;
      } else if (v.length > 3) {
        v = `${v.slice(0, 3)}.${v.slice(3)}`;
      }
      e.target.value = v;
      validateRegisterForm();
    });
  });

  waInputs.forEach(input => {
    if (!input) return;
    input.addEventListener("input", (e) => {
      let v = e.target.value.replace(/\D/g, "");
      if (v.length > 11) v = v.slice(0, 11);
      if (v.length > 6) {
        v = `(${v.slice(0, 2)}) ${v.slice(2, 7)}-${v.slice(7)}`;
      } else if (v.length > 2) {
        v = `(${v.slice(0, 2)}) ${v.slice(2)}`;
      } else if (v.length > 0) {
        v = `(${v}`;
      }
      e.target.value = v;
      validateRegisterForm();
    });
  });

  // Escutar alterações nos campos para validar botões
  const regFields = ["register-name", "register-dob", "register-cpf", "register-whatsapp"];
  regFields.forEach(id => {
    const el = document.getElementById(id);
    if (el) el.addEventListener("input", validateRegisterForm);
  });
}

// --- VALIDAÇÃO DA TELA DE CADASTRO ---
let selectedRegisterRating = 0;

function validateRegisterForm() {
  const name = document.getElementById("register-name").value.trim();
  const dob = document.getElementById("register-dob").value;
  const cpf = document.getElementById("register-cpf").value;
  const whatsapp = document.getElementById("register-whatsapp").value;
  const btn = document.getElementById("btn-submit-register");

  if (!btn) return;

  // Validar Idade (> 16 anos)
  let ageValid = false;
  if (dob) {
    const birthday = new Date(dob);
    const ageDifMs = Date.now() - birthday.getTime();
    const ageDate = new Date(ageDifMs);
    const age = Math.abs(ageDate.getUTCFullYear() - 1970);
    ageValid = age >= 16;
  }

  // CPF completo: 14 caracteres com máscara (000.000.000-00)
  const cpfValid = cpf.length === 14;

  // WhatsApp completo: 15 caracteres com máscara ((00) 90000-0000)
  const waValid = whatsapp.length === 15;

  const formValid = name.length >= 3 && ageValid && cpfValid && waValid && selectedRegisterRating > 0;
  btn.disabled = !formValid;
}

// --- SISTEMA DE TOASTS ---
function showToast(message, type = "success") {
  const container = document.getElementById("toast-container");
  if (!container) return;

  const toast = document.createElement("div");
  toast.className = `toast toast-${type}`;
  
  let iconName = "check-circle";
  if (type === "warning") iconName = "alert-triangle";
  if (type === "error") iconName = "x-octagon";

  toast.innerHTML = `
    <i data-feather="${iconName}"></i>
    <span>${message}</span>
  `;

  container.appendChild(toast);
  feather.replace();

  // Remover após 4s
  setTimeout(() => {
    toast.style.animation = "toastOut 0.3s cubic-bezier(0.4, 0, 0.2, 1) forwards";
    setTimeout(() => toast.remove(), 300);
  }, 3500);
}

// --- CONTROLE DE NAVEGAÇÃO DE ABAS GERAIS ---
function setupTabNavigation() {
  const tabButtons = document.querySelectorAll(".tabs-navigation .tab-btn");
  tabButtons.forEach(btn => {
    btn.addEventListener("click", () => {
      const parent = btn.parentElement;
      // Desativar botões irmãos
      parent.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");

      // Desativar conteúdos de abas correspondentes
      const siblingContents = parent.nextElementSibling ? parent.parentElement.querySelectorAll(".tab-content") : document.querySelectorAll(".tab-content");
      siblingContents.forEach(content => {
        if (content.parentElement === parent.parentElement || content.id.startsWith("tab-")) {
          content.classList.remove("active");
        }
      });

      // Ativar a aba correspondente
      const targetId = btn.getAttribute("data-target");
      const targetContent = document.getElementById(targetId);
      if (targetContent) {
        targetContent.classList.add("active");
        // Trigger para ações específicas da aba
        onTabActivated(targetId);
      }
    });
  });
}

function onTabActivated(tabId) {
  if (tabId === "tab-player-convocacao") {
    renderPlayerConvocationTab();
  } else if (tabId === "tab-player-acompanhamento") {
    renderAcompanhamentoTab();
  } else if (tabId === "tab-player-ranking" || tabId === "tab-manager-athletes") {
    renderRankingTab();
    if (tabId === "tab-manager-athletes") renderManagerAthletesList();
  } else if (tabId === "tab-player-desempenho") {
    renderDesempenhoTab();
  } else if (tabId === "tab-manager-formation") {
    renderManagerCheckinList();
    renderDrawnTeams();
  } else if (tabId === "tab-manager-financeiro") {
    renderManagerFinanceiroTab();
  }
}

// --- EVENT BINDING ---
function bindEvents() {
  // Login e Cadastro links
  document.getElementById("link-goto-register").addEventListener("click", (e) => {
    e.preventDefault();
    document.getElementById("auth-login-form").classList.add("hidden");
    document.getElementById("auth-register-form").classList.remove("hidden");
  });

  document.getElementById("link-goto-login").addEventListener("click", (e) => {
    e.preventDefault();
    document.getElementById("auth-register-form").classList.add("hidden");
    document.getElementById("auth-login-form").classList.remove("hidden");
  });

  // Estrelas de Autoavaliação no Cadastro
  const regStars = document.querySelectorAll("#register-stars-selector .rating-star");
  regStars.forEach(star => {
    star.addEventListener("click", () => {
      const val = parseInt(star.getAttribute("data-value"));
      selectedRegisterRating = val;
      regStars.forEach((s, idx) => {
        if (idx < val) s.classList.add("active");
        else s.classList.remove("active");
      });
      validateRegisterForm();
    });
  });

  // Estrelas de Cadastro/Edição no Modal de Atleta (Gestor)
  const athleteStars = document.querySelectorAll("#athlete-stars-selector .rating-star");
  athleteStars.forEach(star => {
    star.addEventListener("click", () => {
      const val = parseInt(star.getAttribute("data-value"));
      document.getElementById("athlete-stars-selector").dataset.value = val;
      athleteStars.forEach((s, idx) => {
        if (idx < val) s.classList.add("active");
        else s.classList.remove("active");
      });
    });
  });

  // Submissão do Cadastro
  document.getElementById("btn-submit-register").addEventListener("click", handleRegister);

  // Submissão de Login Normal
  document.getElementById("btn-submit-login").addEventListener("click", handleLogin);

  // Login Rápido

    const players = JSON.parse(localStorage.getItem("players"));

  });

  document.getElementById("btn-join-match").addEventListener("click", () => openPaymentModal());

    // Alterna para aba convocação e abre modal de pagamento
    const convTabBtn = document.querySelector(".tabs-navigation .tab-btn[data-target='tab-player-convocacao']");
    if (convTabBtn) convTabBtn.click();
    openPaymentModal();
  });
  document.getElementById("btn-leave-match").addEventListener("click", () => openRemovalModal());

  // Confirmação de Pagamento no Modal
  document.getElementById("btn-confirm-payment-action").addEventListener("click", handleConfirmPayment);
  
  // Mudança do método de pagamento (PIX vs Saldo)
  document.getElementById("pay-method-pix").addEventListener("change", (e) => {
    if (e.target.checked) document.getElementById("pix-qr-area").classList.remove("hidden");
  });
  document.getElementById("pay-method-balance").addEventListener("change", (e) => {
    if (e.target.checked) document.getElementById("pix-qr-area").classList.add("hidden");
  });

  // Confirmação de Remoção no Modal
  document.getElementById("btn-confirm-removal-action").addEventListener("click", handleConfirmRemoval);

  // Ações do Gestor - Atletas
  document.getElementById("athlete-search-input").addEventListener("input", debounce((e) => {
    renderManagerAthletesList(e.target.value);
  }, 300));
  
  document.getElementById("btn-open-add-athlete-modal").addEventListener("click", () => openAthleteModal());
  document.getElementById("btn-save-athlete").addEventListener("click", handleSaveAthlete);

  // Ações do Gestor - Sorteio
  document.getElementById("btn-draw-teams").addEventListener("click", () => {
    openModal("modal-sorteio");
  });
  document.getElementById("btn-execute-sorteio").addEventListener("click", handleExecuteSorteio);

  // Ações do Gestor - Cronômetro Partida
  document.getElementById("btn-timer-toggle").addEventListener("click", toggleLiveTimer);
  document.getElementById("btn-timer-reset").addEventListener("click", resetLiveTimer);
  document.getElementById("btn-finish-match").addEventListener("click", handleFinishMatch);

    m.addEventListener("click", (e) => {
      if (e.target === m) closeModal(m.id);
    });
  });

  setupTabNavigation();
}

    tabManager.classList.add("active");

function renderConvocadosList(peladaId) {
  const convocations = JSON.parse(localStorage.getItem("convocations"));
  const players = JSON.parse(localStorage.getItem("players"));
  const listContainer = document.getElementById("player-convocados-list");
  
  listContainer.innerHTML = "";

  // Filtrar convocações desta pelada
  const matchConvs = convocations.filter(c => c.pelada_id === peladaId);
  
  // Atualizar badge de contagem
  const activeCount = matchConvs.filter(c => c.status === "confirmado").length;
  document.getElementById("convocation-count-badge").textContent = `${activeCount} Confirmados`;

  if (matchConvs.length === 0) {
    listContainer.innerHTML = `
      <div class="empty-state">
        <i data-feather="info" style="width: 24px; height: 24px;"></i>
        <p>Ainda não há convocados para este jogo. Seja o primeiro!</p>
      </div>
    `;
    feather.replace();
    return;
  }

  // Ordenar confirmados primeiro
  matchConvs.sort((a,b) => {
    if (a.status === b.status) return 0;
    return a.status === "confirmado" ? -1 : 1;
  });

  matchConvs.forEach((c) => {
    const player = players.find(p => p.id === c.jogador_id);
    if (!player) return;

    const div = document.createElement("div");
    div.className = "queue-item";
    div.style.borderLeftColor = c.status === "confirmado" ? "var(--secondary)" : "var(--danger)";
    
    let labelGk = player.goleiro ? "🧤 GK" : "";
    let ratingStars = "★".repeat(player.autoavaliacao);
    
    div.innerHTML = `
      <div style="display: flex; align-items: center; gap: 8px;">
        <span class="pos" style="font-size: 14px; width: auto; font-family: 'Inter'; font-weight: normal; color: var(--text-caption);">${labelGk}</span>
        <span style="${c.status === 'cortado' ? 'text-decoration: line-through; opacity: 0.6;' : ''}">${player.nome}</span>
      </div>
      <div style="display: flex; align-items: center; gap: 10px;">
        <span class="player-draft-stars">${ratingStars}</span>
        <span class="badge-status ${c.status}">${c.status}</span>
      </div>
    `;
    listContainer.appendChild(div);
  });
}

// --- CONFIRMAÇÃO DE PAGAMENTO ---
function handleConfirmPayment() {
  const method = document.querySelector('input[name="payment-method"]:checked').value;
  const cost = activeSelectedPelada.valor_convocacao;

  const convocations = JSON.parse(localStorage.getItem("convocations"));
  const players = JSON.parse(localStorage.getItem("players"));
  const transactions = JSON.parse(localStorage.getItem("transactions"));

  // Verificar se o jogador já está confirmado nesta pelada
  const alreadyJoined = convocations.find(c => c.pelada_id === activeSelectedPelada.id && c.jogador_id === currentUser.id && c.status === "confirmado");
  if (alreadyJoined) {
    showToast("Você já confirmou presença para esta pelada!", "warning");
    closeModal("modal-payment");
    return;
  }

  // Função para efetivar
  const executeJoin = () => {
    // 1. Atualizar ou criar convocação
    const existingConv = convocations.find(c => c.pelada_id === activeSelectedPelada.id && c.jogador_id === currentUser.id);
    if (existingConv) {
      existingConv.status = "confirmado";
      existingConv.forma_pagamento = method;
      existingConv.data_convocacao = new Date().toISOString();
    } else {
      convocations.push({
        id: "c_" + Date.now(),
        pelada_id: activeSelectedPelada.id,
        jogador_id: currentUser.id,
        status: "confirmado",
        forma_pagamento: method,
        data_convocacao: new Date().toISOString()
      });
    }

    // 2. Registrar Finanças se for por Saldo Interno (Debita o saldo do jogador)
    const playerInDb = players.find(p => p.id === currentUser.id);
    if (method === "saldo") {
      playerInDb.saldo -= cost;
      
      transactions.push({
        id: "t_" + Date.now(),
        pelada_id: activeSelectedPelada.id,
        jogador_id: currentUser.id,
        tipo: "convocacao",
        descricao: `Convocação debitada em saldo: ${currentUser.nome}`,
        valor: cost,
        sinal: "credito", // Dinheiro entra no caixa geral da pelada
        data: new Date().toISOString()
      });
    } else {
      // Se for por PIX, o saldo do jogador não altera, mas o dinheiro vai pro caixa
      transactions.push({
        id: "t_" + Date.now(),
        pelada_id: activeSelectedPelada.id,
        jogador_id: currentUser.id,
        tipo: "pagamento_pix",
        descricao: `Convocação paga via PIX: ${currentUser.nome}`,
        valor: cost,
        sinal: "credito",
        data: new Date().toISOString()
      });
    }

    localStorage.setItem("convocations", JSON.stringify(convocations));
    localStorage.setItem("players", JSON.stringify(players));
    localStorage.setItem("transactions", JSON.stringify(transactions));

    showToast("Presença confirmada com sucesso!");
    closeModal("modal-payment");
    updatePlayerCardFIFA();
    renderConvocadosList(activeSelectedPelada.id);
  };

  // Simulação de PIX
  if (method === "pix") {
    const confirmBtn = document.getElementById("btn-confirm-payment-action");
    confirmBtn.disabled = true;
    confirmBtn.textContent = "Aguardando aprovação do PIX...";
    
    setTimeout(() => {
      confirmBtn.disabled = false;
      confirmBtn.innerHTML = "Concluir Convocação";
      executeJoin();
    }, 2000);
  } else {
    executeJoin();
  }
}

function handleConfirmRemoval() {
  const convocations = JSON.parse(localStorage.getItem("convocations"));
  const players = JSON.parse(localStorage.getItem("players"));
  const transactions = JSON.parse(localStorage.getItem("transactions"));

  const playerConv = convocations.find(c => c.pelada_id === activeSelectedPelada.id && c.jogador_id === currentUser.id && c.status === "confirmado");
  
  if (!playerConv) return;

  const peladaDateTime = new Date(`${activeSelectedPelada.data}T${activeSelectedPelada.horario}:00`);
  const now = new Date();
  const timeDiffMs = peladaDateTime.getTime() - now.getTime();
  const hoursLeft = timeDiffMs / (1000 * 60 * 60);

  const cost = activeSelectedPelada.valor_convocacao;
  const playerInDb = players.find(p => p.id === currentUser.id);

  if (hoursLeft >= 2) {
    // Estorno automático
    playerConv.status = "cortado";
    playerConv.motivo_remocao = "estorno";
    playerConv.data_remocao = new Date().toISOString();

    // Se pagou em saldo ou pix, estorna para o saldo do jogador
    playerInDb.saldo += cost;

    transactions.push({
      id: "t_" + Date.now(),
      pelada_id: activeSelectedPelada.id,
      jogador_id: currentUser.id,
      tipo: "estorno",
      descricao: `Reembolso de convocação (Prazo OK): ${currentUser.nome}`,
      valor: cost,
      sinal: "debito", // sai do caixa geral
      data: new Date().toISOString()
    });

    showToast("Presença cancelada. Valor estornado ao seu saldo!");
  } else {
    // Perde a taxa -> Vai para o caixa
    playerConv.status = "cortado";
    playerConv.motivo_remocao = "caixa";
    playerConv.data_remocao = new Date().toISOString();

    // Se o pagamento inicial foi em Pix, o saldo dele não muda, mas a pelada fica com a grana.
    // Se foi saldo, já foi debitado, apenas mantemos o débito e transferimos como receita de multa definitiva.
    transactions.push({
      id: "t_" + Date.now(),
      pelada_id: activeSelectedPelada.id,
      jogador_id: currentUser.id,
      tipo: "caixa",
      descricao: `Cancelamento fora do prazo (Taxa retida): ${currentUser.nome}`,
      valor: cost,
      sinal: "credito", // mantido no caixa
      data: new Date().toISOString()
    });

    showToast("Presença cancelada. Taxa retida no caixa da pelada.", "warning");
  }

  localStorage.setItem("convocations", JSON.stringify(convocations));
  localStorage.setItem("players", JSON.stringify(players));
  localStorage.setItem("transactions", JSON.stringify(transactions));

  closeModal("modal-removal");
  updatePlayerCardFIFA();
  renderConvocadosList(activeSelectedPelada.id);
}

// --- POLLING SIMULADO DE ACOMPANHAMENTO (A cada 10s) ---
let pollingActive = false;
function startSimulatedPolling() {
  if (pollingActive) return;
  pollingActive = true;

  setInterval(() => {
    // Apenas atualiza se o timer estiver correndo
    if (liveMatch.isPlaying) {
      // Simula uma probabilidade sutil de gols
      const rand = Math.random();
      if (rand > 0.85) {
        if (Math.random() > 0.5) {
          liveMatch.scoreA++;
          showToast(`GOL do ${liveMatch.teamA}!`);
        } else {
          liveMatch.scoreB++;
          showToast(`GOL do ${liveMatch.teamB}!`);
        }
        
        // Sincronizar com os campos do Gestor caso estejam visíveis
        const gestorScoreA = document.getElementById("match-control-score-a");
        const gestorScoreB = document.getElementById("match-control-score-b");
        if (gestorScoreA) gestorScoreA.textContent = liveMatch.scoreA;
        if (gestorScoreB) gestorScoreB.textContent = liveMatch.scoreB;

        updateAcompanhamentoUI();
      }
    }
  }, 10000);
}

function handleSaveAthlete() {
  const id = document.getElementById("athlete-edit-id").value;
  const name = document.getElementById("athlete-name").value.trim();
  const dob = document.getElementById("athlete-dob").value;
  const cpf = document.getElementById("athlete-cpf").value;
  const whatsapp = document.getElementById("athlete-whatsapp").value;
  const isGk = document.getElementById("athlete-is-gk").checked;
  const rating = parseInt(document.getElementById("athlete-stars-selector").dataset.value) || 0;

  if (!name || !dob || !cpf || !whatsapp || rating === 0) {
    showToast("Por favor, preencha todos os campos do atleta.", "error");
    return;
  }

  const players = JSON.parse(localStorage.getItem("players"));

  if (id) {
    // Atualizar
    const p = players.find(x => x.id === id);
    p.nome = name;
    p.data_nascimento = dob;
    p.cpf = cpf;
    p.whatsapp = whatsapp;
    p.goleiro = isGk;
    p.autoavaliacao = rating;
    showToast("Atleta atualizado!");
  } else {
    // Criar
    if (players.some(x => x.cpf === cpf)) {
      showToast("Já existe um atleta cadastrado com este CPF.", "error");
      return;
    }

    players.push({
      id: "p_" + Date.now(),
      nome: name,
      data_nascimento: dob,
      cpf: cpf,
      whatsapp: whatsapp,
      goleiro: isGk,
      autoavaliacao: rating,
      ativo: true,
      saldo: 0.00,
      gols: 0,
      partidas: 0,
      avaliacao_media: rating
    });
    showToast("Novo atleta cadastrado com sucesso!");
  }

  localStorage.setItem("players", JSON.stringify(players));
  closeModal("modal-athlete");
  renderManagerAthletesList();
}

}

// --- ALGORITMO SNAKE DRAFT DE SORTEIO ---
function handleExecuteSorteio() {
  const type = document.querySelector('input[name="sorteio-type"]:checked').value;
  closeModal("modal-sorteio");

  const players = JSON.parse(localStorage.getItem("players"));
  const configs = JSON.parse(localStorage.getItem("configs"));
  const config = configs.find(c => c.grupo_id === currentGroup.id) || { jogadores_por_time: 6, quantidade_times: 3 };

  const playersPerTeam = config.jogadores_por_time;
  let qtyTeams = config.quantidade_times;

  // Filtrar jogadores da lista de presentes do banco de dados
  const activePresent = players.filter(p => presentPlayers.includes(p.id));

  if (activePresent.length < 2) {
    showToast("Número insuficiente de jogadores presentes para o sorteio.", "error");
    return;
  }

  // Separar goleiros e jogadores de linha
  const gkList = activePresent.filter(p => p.goleiro);
  const fieldList = activePresent.filter(p => !p.goleiro);

  // Inicializar times
  drawnTeams = [];
  const teamColors = ["#2196F3", "#FFC107", "#FF1744", "#00C853", "#FF6D00", "#9C27B0", "#E91E63", "#00BCD4", "#795548", "#607D8B"];

  for (let i = 0; i < qtyTeams; i++) {
    drawnTeams.push({
      id: `team_${i+1}`,
      nome: `Time ${getColoName(i)}`,
      cor: teamColors[i] || "#777",
      players: []
    });
  }

  // 1. DISTRIBUIR GOLEIROS (1 por time aleatoriamente)
  const shuffledGks = shuffleArray([...gkList]);
  shuffledGks.forEach((gk, idx) => {
    if (idx < qtyTeams) {
      drawnTeams[idx].players.push(gk);
    } else {
      // Goleiros extras vão para a lista de campo ou reserva
      fieldList.push(gk);
    }
  });

  // 2. ORDENAR JOGADORES DE CAMPO POR ESTRELAS (Decrescente)
  // E aplicar uma randomização sutil: embaralhar ligeiramente jogadores de mesmo nível
  let sortedField = [...fieldList].sort((a, b) => {
    if (b.autoavaliacao === a.autoavaliacao) {
      return Math.random() - 0.5; // Embaralha jogadores de mesma estrela
    }
    return b.autoavaliacao - a.autoavaliacao;
  });

  // Randomização de ±1 posição na serpentina
  // Damos uma leve perturbada na lista ordenada para evitar determinismo puro
  for (let i = 0; i < sortedField.length - 1; i++) {
    if (Math.random() < 0.25) { // 25% de chance de trocar vizinho
      const temp = sortedField[i];
      sortedField[i] = sortedField[i + 1];
      sortedField[i + 1] = temp;
    }
  }

  // Se a opção for Sorteio B (Necessários), cortamos a lista no tamanho máximo dos times
  let maxRequired = qtyTeams * playersPerTeam;
  let remainingPlayers = [];

  if (type === "necessarios" && sortedField.length + gkList.length > maxRequired) {
    const maxFieldNeeded = maxRequired - Math.min(gkList.length, qtyTeams);
    remainingPlayers = sortedField.slice(maxFieldNeeded);
    sortedField = sortedField.slice(0, maxFieldNeeded);
  }

  // 3. DISTRIBUIÇÃO EM SERPENTINA (Snake Draft)
  let direction = 1; // 1 = A->D, -1 = D->A
  let teamIdx = 0;

  sortedField.forEach((player) => {
    // Acha o time menos cheio que não seja goleiro extra se já tiver GK
    drawnTeams[teamIdx].players.push(player);

    // Mover o índice
    teamIdx += direction;
    if (teamIdx >= qtyTeams) {
      teamIdx = qtyTeams - 1;
      direction = -1; // Inverte
    } else if (teamIdx < 0) {
      teamIdx = 0;
      direction = 1; // Inverte
    }
  });

  // 4. VALIDAÇÃO TÉCNICA (Média de estrelas deve variar no máx 0.5)
  // Se falhar a validação, faremos pequenas correções (SWAP) para equilibrar
  balanceDrawnTeams(playersPerTeam);

  // Guardar times no LocalStorage
  localStorage.setItem("teams", JSON.stringify(drawnTeams));

  // Alimentar a fila de espera
  waitingQueue = [];
  drawnTeams.forEach((t, idx) => {
    if (idx >= 2) { // 2 primeiros estão em campo, outros na fila
      waitingQueue.push(t.nome);
    }
  });

  // Definir partida ao vivo inicial
  if (drawnTeams.length >= 2) {
    liveMatch.teamA = drawnTeams[0].nome;
    liveMatch.teamB = drawnTeams[1].nome;
    liveMatch.scoreA = 0;
    liveMatch.scoreB = 0;
    liveMatch.isPlaying = false;
    
    // Atualizar controle no Gestor
    document.getElementById("match-control-team-a").textContent = liveMatch.teamA;
    document.getElementById("match-control-team-b").textContent = liveMatch.teamB;
    document.getElementById("match-control-score-a").textContent = "0";
    document.getElementById("match-control-score-b").textContent = "0";
  }

  showToast("Times sorteados com equilíbrio técnico equilibrado!");
  renderDrawnTeams();
  updateAcompanhamentoUI();
}

function getColoName(idx) {
  const names = ["Azul", "Amarelo", "Vermelho", "Verde", "Laranja", "Roxo", "Rosa", "Ciano", "Marrom", "Cinza"];
  return names[idx] || `Equipe ${idx+1}`;
}

// Algoritmo simples de balanceamento
function balanceDrawnTeams(playersPerTeam) {
  let attempts = 0;
  const getAverage = (team) => {
    if (team.players.length === 0) return 0;
    return team.players.reduce((sum, p) => sum + p.autoavaliacao, 0) / team.players.length;
  };

  while (attempts < 15) {
    let minTeam = drawnTeams[0];
    let maxTeam = drawnTeams[0];

    drawnTeams.forEach(t => {
      if (getAverage(t) < getAverage(minTeam)) minTeam = t;
      if (getAverage(t) > getAverage(maxTeam)) maxTeam = t;
    });

    const diff = getAverage(maxTeam) - getAverage(minTeam);
    if (diff <= 0.5) break; // Sucesso! Diferença menor ou igual a 0.5 estrelas

    // Tentar trocar um jogador de alta avaliação do maxTeam com um de baixa avaliação do minTeam (sem trocar goleiros)
    let swapped = false;
    const maxPlayers = maxTeam.players.filter(p => !p.goleiro);
    const minPlayers = minTeam.players.filter(p => !p.goleiro);

    for (let pMax of maxPlayers) {
      for (let pMin of minPlayers) {
        if (pMax.autoavaliacao > pMin.autoavaliacao) {
          // Efetuar troca temporária e testar se melhora
          const idxMax = maxTeam.players.indexOf(pMax);
          const idxMin = minTeam.players.indexOf(pMin);
          
          maxTeam.players[idxMax] = pMin;
          minTeam.players[idxMin] = pMax;

          const newDiff = getAverage(maxTeam) - getAverage(minTeam);
          if (newDiff < diff) {
            swapped = true;
            break;
          } else {
            // Desfaz
            maxTeam.players[idxMax] = pMax;
            minTeam.players[idxMin] = pMin;
          }
        }
      }
      if (swapped) break;
    }

    if (!swapped) break; // Nenhuma troca viável melhorou
    attempts++;
  }
}

// Helper shuffle
function shuffleArray(array) {
  for (let i = array.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [array[i], array[j]] = [array[j], array[i]];
  }
  return array;
}

    const card = document.createElement("div");

    card.id = `card-team-${team.id}`;
    card.setAttribute("ondragover", "allowDrop(event)");
    card.setAttribute("ondragleave", "dragLeave(event)");
    card.setAttribute("ondrop", `drop(event, '${team.id}')`);

    card.innerHTML = `
      <div class="team-draft-header">
        <input type="text" class="team-draft-title-input" value="${team.nome}" onchange="renameTeam('${team.id}', this.value)">
        <span style="font-size:11px; font-weight:bold; color:var(--text-caption);">Avg: ⭐${avg}</span>
      </div>
      <div class="team-draft-players" id="players-list-${team.id}">
        <!-- Jogadores do Time -->
      </div>
    `;

    const playersList = card.querySelector(`#players-list-${team.id}`);
    team.players.forEach(p => {
      const pDiv = document.createElement("div");
      pDiv.className = "player-draft-item";
      pDiv.draggable = true;
      pDiv.setAttribute("ondragstart", `drag(event, '${p.id}', '${team.id}')`);
      
      pDiv.innerHTML = `
        <span class="player-draft-name">
          ${p.nome} ${p.goleiro ? '🧤' : ''}
        </span>
        <span class="player-draft-stars">${"★".repeat(p.autoavaliacao)}</span>
      `;
      playersList.appendChild(pDiv);
    });

    container.appendChild(card);
  });
}

// DRAG AND DROP HANDLERS NATIVER
let draggedPlayerId = null;
let draggedFromTeamId = null;

window.drag = function(ev, playerId, teamId) {
  draggedPlayerId = playerId;
  draggedFromTeamId = teamId;
  ev.dataTransfer.setData("text", playerId);
};

window.allowDrop = function(ev) {
  ev.preventDefault();
  const card = ev.currentTarget;
  card.classList.add("dragover");
};

window.dragLeave = function(ev) {
  const card = ev.currentTarget;
  card.classList.remove("dragover");
};

window.drop = function(ev, targetTeamId) {
  ev.preventDefault();
  const card = ev.currentTarget;
  card.classList.remove("dragover");

  if (targetTeamId === draggedFromTeamId) return;

  const sourceTeam = drawnTeams.find(t => t.id === draggedFromTeamId);
  const targetTeam = drawnTeams.find(t => t.id === targetTeamId);
  const player = sourceTeam.players.find(p => p.id === draggedPlayerId);

  // Mover
  sourceTeam.players = sourceTeam.players.filter(p => p.id !== draggedPlayerId);
  targetTeam.players.push(player);

  // Guardar no banco
  localStorage.setItem("teams", JSON.stringify(drawnTeams));

  // Recalcular médias e re-renderizar
  renderDrawnTeams();
  showToast(`${player.nome} movido para ${targetTeam.nome}!`);
};

window.renameTeam = function(teamId, newName) {
  const team = drawnTeams.find(t => t.id === teamId);
  if (team && newName.trim()) {
    const oldName = team.nome;
    team.nome = newName.trim();
    localStorage.setItem("teams", JSON.stringify(drawnTeams));
    
    // Atualizar na fila de espera ou placar vivo
    if (waitingQueue.includes(oldName)) {
      waitingQueue = waitingQueue.map(n => n === oldName ? team.nome : n);
    }
    if (liveMatch.teamA === oldName) {
      liveMatch.teamA = team.nome;
      document.getElementById("match-control-team-a").textContent = team.nome;
    }
    if (liveMatch.teamB === oldName) {
      liveMatch.teamB = team.nome;
      document.getElementById("match-control-team-b").textContent = team.nome;
    }
    
    updateAcompanhamentoUI();
    showToast(`Time renomeado para ${team.nome}`);
  }
};

// --- CONTROLE DE JOGO E CRONÔMETRO AO VIVO ---
function toggleLiveTimer() {
  const btn = document.getElementById("btn-timer-toggle");
  
  if (liveMatch.isPlaying) {
    // Pausar
    clearInterval(timerInterval);
    liveMatch.isPlaying = false;
    btn.textContent = "Retomar";
    btn.className = "btn btn-sm btn-primary";
    showToast("Cronômetro pausado!");
  } else {
    // Iniciar
    liveMatch.isPlaying = true;
    btn.textContent = "Pausar";
    btn.className = "btn btn-sm btn-outline";
    
    timerInterval = setInterval(() => {
      timerSeconds++;
      updateTimerDisplay();
    }, 1000);
    showToast("Partida iniciada!");
  }
}

function resetLiveTimer() {
  clearInterval(timerInterval);
  liveMatch.isPlaying = false;
  timerSeconds = 0;
  
  document.getElementById("btn-timer-toggle").textContent = "Iniciar";
  document.getElementById("btn-timer-toggle").className = "btn btn-sm btn-primary";
  
  updateTimerDisplay();
  showToast("Cronômetro zerado.");
}

function updateTimerDisplay() {
  const min = Math.floor(timerSeconds / 60).toString().padStart(2, "0");
  const sec = (timerSeconds % 60).toString().padStart(2, "0");
  const text = `${min}:${sec}`;

  // Atualizar tela do gestor
  document.getElementById("match-control-timer").textContent = text;
  
  // Atualizar tela do jogador (Acompanhamento)
  const userTimer = document.getElementById("live-match-timer");
  const userProgressBar = document.getElementById("live-match-timer-progress");

  if (userTimer) userTimer.textContent = text;
  if (userProgressBar) {
    // Simula barra de 10 minutos (600s)
    const pct = Math.min((timerSeconds / 600) * 100, 100);
    userProgressBar.style.width = `${pct}%`;
  }
}

window.updateLiveScore = function(team, diff) {
  if (team === "a") {
    liveMatch.scoreA = Math.max(0, liveMatch.scoreA + diff);
    document.getElementById("match-control-score-a").textContent = liveMatch.scoreA;
  } else {
    liveMatch.scoreB = Math.max(0, liveMatch.scoreB + diff);
    document.getElementById("match-control-score-b").textContent = liveMatch.scoreB;
  }
  updateAcompanhamentoUI();
};

function handleFinishMatch() {
  if (drawnTeams.length < 2) {
    showToast("Não há times suficientes configurados para finalizar partidas.", "error");
    return;
  }

  const scoreA = liveMatch.scoreA;
  const scoreB = liveMatch.scoreB;

  showToast(`Partida Encerrada! Resultado: ${liveMatch.teamA} ${scoreA} x ${scoreB} ${liveMatch.teamB}`);

  // Aplicar regras de vitória/empate e rotacionar a fila de espera
  const configs = JSON.parse(localStorage.getItem("configs"));
  const config = configs.find(c => c.grupo_id === currentGroup.id) || { criterio_empate: "ambos_permanecem", regra_saida: "final_fila" };

  let winnerName = "";
  let loserName = "";
  let isTie = scoreA === scoreB;

  if (!isTie) {
    winnerName = scoreA > scoreB ? liveMatch.teamA : liveMatch.teamB;
    loserName = scoreA > scoreB ? liveMatch.teamB : liveMatch.teamA;
  }

  // Lógica de Rotatividade de Times baseada nas configurações
  if (isTie) {
    // Empate
    const rule = config.criterio_empate;
    
    if (rule === "saem_ambos") {
      // Ambos os times saem, entram dois novos da fila
      if (waitingQueue.length >= 2) {
        const nextA = waitingQueue.shift();
        const nextB = waitingQueue.shift();
        waitingQueue.push(liveMatch.teamA);
        waitingQueue.push(liveMatch.teamB);
        
        liveMatch.teamA = nextA;
        liveMatch.teamB = nextB;
      }
    } else if (rule === "time_entrando") {
      // O time que estava no final da fila permanece (não aplicável facilmente a menos que o entrante seja o vencedor)
      // Rotaciona um time derrotado (geralmente o time que está em campo há mais tempo)
      const next = waitingQueue.shift();
      waitingQueue.push(liveMatch.teamB); // B sai
      liveMatch.teamB = next;
    } else {
      // default: ambos permanecem ou gestor decide. Mantemos ambos.
      showToast("Pela regra, ambos os times permanecem em campo.");
    }
  } else {
    // Vitória de alguém
    // O perdedor vai para onde a regra de saída mandar
    const exitRule = config.regra_saida;

    if (waitingQueue.length > 0) {
      const nextTeam = waitingQueue.shift(); // Próximo time entra
      
      if (exitRule === "final_fila") {
        waitingQueue.push(loserName); // Perdedor vai pro final da fila
      } else {
        // fora_1_jogo: vai pro final da fila também na simulação simplificada
        waitingQueue.push(loserName);
      }

      // Vencedor continua em campo contra o entrante
      if (winnerName === liveMatch.teamA) {
        liveMatch.teamB = nextTeam;
      } else {
        liveMatch.teamA = nextTeam;
      }
    }
  }

  // Zerar placar e tempo
  liveMatch.scoreA = 0;
  liveMatch.scoreB = 0;
  resetLiveTimer();

  // Sincronizar UI
  document.getElementById("match-control-team-a").textContent = liveMatch.teamA;
  document.getElementById("match-control-team-b").textContent = liveMatch.teamB;
  document.getElementById("match-control-score-a").textContent = "0";
  document.getElementById("match-control-score-b").textContent = "0";

  updateAcompanhamentoUI();
}

// ==================== MÓDULO GESTOR — FINANCEIRO ====================
function renderManagerFinanceiroTab() {
  const transactions = JSON.parse(localStorage.getItem("transactions"));
  const players = JSON.parse(localStorage.getItem("players"));

  // Calcular Balanço Geral
  let receipts = 0;
  let expenses = 0;

  transactions.forEach(t => {
    // Apenas transações do grupo atual
    // Como no mock simplificamos sem multi-grupo rígido nas transações, filtramos todas do localStorage
    if (t.sinal === "credito") receipts += t.valor;
    else expenses += t.valor;
  });

  const net = receipts - expenses;

  document.getElementById("finances-receipts").textContent = `R$ ${receipts.toFixed(2).replace(".", ",")}`;
  document.getElementById("finances-expenses").textContent = `R$ ${expenses.toFixed(2).replace(".", ",")}`;
  
  const netEl = document.getElementById("finances-net");
  netEl.textContent = `R$ ${net.toFixed(2).replace(".", ",")}`;
  netEl.style.color = net >= 0 ? "var(--success)" : "var(--danger)";

  // Renderizar Lista de Transações (Histórico)
  const transBody = document.getElementById("finances-transactions-body");
  transBody.innerHTML = "";

  if (transactions.length === 0) {
    transBody.innerHTML = `<tr><td colspan="3" style="text-align:center; color:var(--text-caption);">Nenhum lançamento.</td></tr>`;
  } else {
    // Mostrar últimos 10
    [...transactions].reverse().slice(0, 10).forEach(t => {
      const tr = document.createElement("tr");
      const dateFormatted = new Date(t.data).toLocaleDateString("pt-BR");
      const valColor = t.sinal === "credito" ? "var(--success)" : "var(--danger)";
      const sign = t.sinal === "credito" ? "+" : "-";

      tr.innerHTML = `
        <td>${dateFormatted}</td>
        <td>
          <span style="font-weight: 500; font-size:13px; display:block;">${t.descricao}</span>
          <span style="font-size:11px; color:var(--text-caption);">${t.tipo}</span>
        </td>
        <td style="text-align: right; font-weight: bold; color: ${valColor};">${sign} R$ ${t.valor.toFixed(2)}</td>
      `;
      transBody.appendChild(tr);
    });
  }

  // Renderizar Balanço de Saldos dos Jogadores
  const balBody = document.getElementById("finances-athlete-balances-body");
  balBody.innerHTML = "";

  // Ordenar por maior dívida (saldos mais negativos primeiro)
  const sortedPlayers = [...players].sort((a,b) => a.saldo - b.saldo);

  sortedPlayers.forEach(p => {
    const tr = document.createElement("tr");
    const balanceColor = p.saldo >= 0 ? "var(--success)" : "var(--danger)";
    
    // Ação Cobrar se for negativo
    let btnCobrar = "";
    if (p.saldo < 0) {
      const whatsMsg = window.encodeURIComponent(`Olá ${p.nome.split(" ")[0]}! Vimos no Pelada Pro que seu saldo está em R$ ${p.saldo.toFixed(2).replace(".", ",")}. Quando puder, faz o acerto via Pix. Valeu!`);
      const linkWhats = `https://api.whatsapp.com/send?phone=55${p.whatsapp.replace(/\D/g, "")}&text=${whatsMsg}`;
      btnCobrar = `
        <a href="${linkWhats}" target="_blank" class="btn btn-sm btn-outline" style="text-decoration:none; padding:4px 8px; border-color:var(--accent); color:var(--accent); font-size:11px; height:24px;">Cobrar</a>
      `;
    }

    tr.innerHTML = `
      <td style="font-weight: 500;">${p.nome} ${p.goleiro ? '🧤' : ''}</td>
      <td style="text-align: right; font-weight: bold; color: ${balanceColor};">R$ ${p.saldo.toFixed(2).replace(".", ",")}</td>
      <td style="text-align: center;">
        <div style="display:flex; gap:6px; justify-content:center;">
          ${btnCobrar}
          <button class="btn btn-sm btn-secondary" onclick="manualFinanceSettlement('${p.id}')" style="font-size:11px; padding:4px 8px; height:24px;">Ajustar</button>
        </div>
      </td>
    `;
    balBody.appendChild(tr);
  });
}

function handleSaveExpense() {
  const desc = document.getElementById("expense-description").value.trim();
  const val = parseFloat(document.getElementById("expense-value").value);
  const cat = document.getElementById("expense-category").value;

  if (!desc || isNaN(val) || val <= 0) {
    showToast("Preencha a descrição e o valor da despesa corretamente.", "error");
    return;
  }

  const transactions = JSON.parse(localStorage.getItem("transactions"));
  transactions.push({
    id: "t_" + Date.now(),
    pelada_id: null,
    jogador_id: null,
    tipo: "despesa",
    descricao: desc,
    valor: val,
    sinal: "debito",
    data: new Date().toISOString(),
    categoria: cat
  });

  localStorage.setItem("transactions", JSON.stringify(transactions));
  
  // Limpar campos
  document.getElementById("expense-description").value = "";
  document.getElementById("expense-value").value = "";

  closeModal("modal-expense");
  showToast("Despesa registrada com sucesso!");
  renderManagerFinanceiroTab();
}

window.manualFinanceSettlement = function(playerId) {
  const players = JSON.parse(localStorage.getItem("players"));
  const p = players.find(x => x.id === playerId);
  if (!p) return;

  const inputAmount = window.prompt(`Ajuste financeiro para ${p.nome}.\nDigite um valor positivo para adicionar créditos, ou negativo para registrar débitos:`, "0");
  const amt = parseFloat(inputAmount);

  if (isNaN(amt) || amt === 0) return;

  p.saldo += amt;

  // Registrar lançamento manual
  const transactions = JSON.parse(localStorage.getItem("transactions"));
  transactions.push({
    id: "t_" + Date.now(),
    pelada_id: null,
    jogador_id: p.id,
    tipo: "acerto",
    descricao: `Acerto financeiro manual: ${p.nome}`,
    valor: Math.abs(amt),
    sinal: amt > 0 ? "credito" : "debito",
    data: new Date().toISOString()
  });

  localStorage.setItem("players", JSON.stringify(players));
  localStorage.setItem("transactions", JSON.stringify(transactions));

  showToast(`Ajuste de R$ ${amt.toFixed(2)} realizado.`);
  renderManagerFinanceiroTab();
};

// --- FUNÇÃO DE DEBOUNCE PARA BUSCA ---
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}
